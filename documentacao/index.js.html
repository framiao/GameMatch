/** Cache de memória para evitar sobrecarga da API do SteamSpy */
const cacheMultiplayer = {};

/**
 * Verifica se um jogo tem a tag de "Multiplayer" ou "Co-op" usando o SteamSpy.
 * Utiliza sistema de cache para respostas instantâneas em jogos já verificados.
 * 
 * @param {string|number} appId - O ID oficial do jogo na Steam.
 * @returns {Promise<boolean>} Retorna true se for multiplayer, false caso contrário.
 */
async function isMultiplayerGame(appId) {
    if (cacheMultiplayer[appId] !== undefined) {
        return cacheMultiplayer[appId];
    }

    try {
        const resposta = await fetch(`https://steamspy.com/api.php?request=appdetails&appid=${appId}`);
        const dados = await resposta.json();
        
        let temMultiplayer = false;
        if (dados.tags) {
            const tagsDoJogo = Object.keys(dados.tags).map(t => t.toLowerCase());
            temMultiplayer = tagsDoJogo.some(tag => 
                tag.includes('multiplayer') || tag.includes('co-op') || 
                tag.includes('coop') || tag === 'split screen'
            );
        }

        cacheMultiplayer[appId] = temMultiplayer;
        return temMultiplayer;
    } catch (erro) {
        console.error(`Erro ao verificar tag do jogo ${appId}:`, erro);
        return true; // Fallback de segurança para não bloquear jogos válidos
    }
}

/**
 * Busca o Nickname oficial e a foto de perfil em alta resolução.
 * 
 * @param {string} playerId - O SteamID64 ou ID de convidado.
 * @param {boolean} isGuest - Flag que define se o jogador é convidado.
 * @param {string} fallbackName - Nome a utilizar caso a API falhe ou seja convidado.
 * @returns {Promise<{name: string, avatar: string}>} Objeto com os detalhes visuais do jogador.
 */
async function getPlayerDetails(playerId, isGuest, fallbackName) {
    if (isGuest) {
        return { name: fallbackName, avatar: `https://api.dicebear.com/7.x/bottts/svg?seed=${fallbackName}` };
    }
    try {
        const url = `http://api.steampowered.com/ISteamUser/GetPlayerSummaries/v0002/?key=${STEAM_API_KEY}&steamids=${playerId}`;
        const resposta = await axios.get(url);
        const player = resposta.data.response.players[0];
        if (player) {
            return { name: player.personaname, avatar: player.avatarfull };
        }
    } catch (erro) {
        console.error('Erro ao buscar perfil na Steam:', erro.message);
    }
    return { name: fallbackName, avatar: 'https://avatars.githubusercontent.com/u/0?v=4' };
}

/** Remove um jogador da sala. Elimina a sala se ficar vazia, ou aborta o jogo se a decorrer. */
app.post('/api/sessions/:roomCode/leave', async (req, res) => {
    try {
        const { playerId } = req.body;
        const sessao = await Session.findOne({ roomCode: req.params.roomCode.toUpperCase() });
        
        if (!sessao) return res.status(404).json({ erro: 'Sala não encontrada.' });

        sessao.players = sessao.players.filter(p => p.id !== playerId);

        if (sessao.players.length === 0) {
            await Session.deleteOne({ _id: sessao._id });
            return res.json({ mensagem: 'Sala eliminada.' });
        }

        // Passa a coroa de Host se quem saiu for o Host
        if (sessao.hostId === playerId && sessao.players.length > 0) {
            sessao.hostId = sessao.players[0].id;
        }

        // Aborta o jogo se alguém desistir a meio
        if (sessao.status === 'playing' && sessao.players.length < 2) {
            sessao.status = 'aborted';
            sessao.matches = [];
            sessao.players.forEach(p => { p.finished = false; p.likes = []; });
        }

        await sessao.save();
        res.json({ mensagem: 'Saída com sucesso.' });
    } catch (erro) {
        console.error('Erro ao sair da sala:', erro);
        res.status(500).json({ erro: 'Falha ao remover jogador.' });
    }
});

/** Computa a intersecção de todas as bibliotecas Steam dos jogadores da Sala. */
app.get('/api/sessions/:roomCode/games', async (req, res) => {
    try {
        const sessao = await Session.findOne({ roomCode: req.params.roomCode.toUpperCase() });
        if (!sessao) return res.status(404).json({ erro: 'Sala não encontrada.' });

        const steamPlayers = sessao.players.filter(p => !p.isGuest);
        let jogosComuns = [];

        if (steamPlayers.length > 0) {
            const promessasJogos = steamPlayers.map(async (player) => {
                const url = `http://api.steampowered.com/IPlayerService/GetOwnedGames/v0001/?key=${STEAM_API_KEY}&steamid=${player.id}&format=json&include_appinfo=true`;
                const resposta = await axios.get(url);
                return resposta.data.response.games || [];
            });

            const listasDeJogos = await Promise.all(promessasJogos);

            if (!listasDeJogos.some(lista => lista.length === 0)) {
                jogosComuns = listasDeJogos[0];
                // Lógica de Intersecção
                for (let i = 1; i < listasDeJogos.length; i++) {
                    const idsDesseJogador = new Set(listasDeJogos[i].map(j => j.appid));
                    jogosComuns = jogosComuns.filter(jogo => idsDesseJogador.has(jogo.appid));
                }

                // Filtro de Multiplayer (SteamSpy)
                const jogosFiltrados = [];
                for (const jogo of jogosComuns) {
                    const isMulti = await isMultiplayerGame(jogo.appid);
                    if (isMulti) {
                        jogosFiltrados.push(jogo);
                    }
                }
                jogosComuns = jogosFiltrados;
                
                // Formatação final
                jogosComuns = jogosComuns.map(jogo => ({
                    id: String(jogo.appid),
                    title: jogo.name,
                    image: `https://steamcdn-a.akamaihd.net/steam/apps/${jogo.appid}/library_600x900.jpg`,
                    tags: ['PC', 'Steam']
                }));
            }
        }

        const listaFinal = [...jogosComuns, ...sessao.customGames];
        const listaFinalSemDuplicados = Array.from(new Map(listaFinal.map(jogo => [jogo.id, jogo])).values());

        res.json({ 
            total_jogos: listaFinalSemDuplicados.length, 
            jogos: listaFinalSemDuplicados 
        });

    } catch (erro) {
        console.error('Erro ao calcular jogos:', erro);
        res.status(500).json({ erro: 'Falha ao buscar os jogos.' });
    }
});

/** Regista um "Gosto/Heart" de um jogador e avalia instantaneamente se houve Match Global. */
app.post('/api/sessions/:roomCode/like', async (req, res) => {
    try {
        const { playerId, gameId } = req.body;
        const roomCode = req.params.roomCode.toUpperCase();

        const sessao = await Session.findOne({ roomCode });
        if (!sessao) return res.status(404).json({ erro: 'Sala não encontrada.' });

        const jogador = sessao.players.find(p => p.id === playerId);
        if (!jogador) return res.status(404).json({ erro: 'Jogador não encontrado na sala.' });

        if (!jogador.likes.includes(gameId)) {
            jogador.likes.push(gameId);
            await sessao.save(); 
        }

        // Verifica se TODOS na sala deram like neste jogo exato
        const isMatch = sessao.players.every(p => p.likes.includes(gameId));

        if (isMatch) {
            if (!sessao.matches.includes(gameId)) {
                sessao.matches.push(gameId);
                await sessao.save();
            }
            return res.json({ isMatch: true, message: 'TEMOS MATCH!', gameId: gameId });
        }

        res.json({ isMatch: false, message: 'Like registado. A aguardar outros.' });

    } catch (erro) {
        console.error('Erro ao registar like:', erro);
        res.status(500).json({ erro: 'Falha ao registar o like.' });
    }
});

/** Assinala que um jogador terminou todas as suas cartas. Declara 'no_match' se todos terminarem sem sucesso. */
app.post('/api/sessions/:roomCode/finish', async (req, res) => {
    try {
        const { playerId } = req.body;
        const sessao = await Session.findOne({ roomCode: req.params.roomCode.toUpperCase() });
        if (!sessao) return res.status(404).json({ erro: 'Sala não encontrada.' });

        const jogador = sessao.players.find(p => p.id === playerId);
        if (jogador) {
            jogador.finished = true;
            await sessao.save();
        }

        const todosTerminaram = sessao.players.every(p => p.finished);
        if (todosTerminaram && sessao.matches.length === 0) {
            sessao.status = 'no_match';
            await sessao.save();
        }

        res.json({ sucesso: true });
    } catch (erro) {
        res.status(500).json({ erro: 'Falha ao terminar.' });
    }
});
